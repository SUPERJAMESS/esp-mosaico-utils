cmake_minimum_required(VERSION 3.16)
get_filename_component(MOSAICO_UTILS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(STRESS_SAMPLE "A" CACHE STRING "Frozen flash stress sample: A, A2, B or C")
if(NOT STRESS_SAMPLE MATCHES "^(A|A2|B|C)$")
    message(FATAL_ERROR "Unknown flash stress sample: ${STRESS_SAMPLE}")
endif()
if(NOT SDKCONFIG)
    set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
endif()
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
if(STRESS_SAMPLE STREQUAL "C")
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.resources.defaults")
endif()
set(EXTRA_COMPONENT_DIRS "${MOSAICO_UTILS_ROOT}/esp-mosaico-vibe-mode/components/esp_mosaico_app_recovery")
include("${MOSAICO_UTILS_ROOT}/esp-mosaico-vibe-mode/cmake/mosaico_idf_project.cmake")
project(flash_stress VERSION 1.0.0)
